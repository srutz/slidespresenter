
# 6.5 React mit TypeScript / Typisierung verschiedener Hooks mit TypeScript

## Typisierung von State

- State kann inline oder als Interface oder Type-Alias definiert werden.

```typescript
function Counter() {
    const [state, setState] = useState<{count: number}>({ count: 0 });

    return (
        <div>
            <button onClick={() => setState({ count: state.count + 1 })}>
                Count: {state.count}
            </button>
        </div>
    )
}
```

## Typisierung von useRef

- useRef kann inline oder als Interface oder Type-Alias definiert werden.

```typescript
function Counter() {
    const countRef = useRef<{current: number}>({ current: 0 });

    return (
        <div>
            <button onClick={() => countRef.current++}>
                Count: {countRef.current}
            </button>
        </div>
    )
}
```

## Typisierung von useContext

- useContext kann inline oder als Interface oder Type-Alias definiert werden.

```typescript
interface ThemeContextType {
    theme: string;
    setTheme: (theme: string) => void;
}

const ThemeContext = createContext<ThemeContextType>({
    theme: 'light',
    setTheme: () => {}
})

function App() {
    const { theme, setTheme } = useContext(ThemeContext);

    return (
        <div>
            <button onClick={() => setTheme('dark')}>
                Toggle Theme
            </button>
            <div>
                Current Theme: {theme}
            </div>
        </div>
    )
}
```
