
# 11.6 Formulare und Validierung / Fehlermeldungen anzeigen

## Mehrere Fehlermeldungen anzeigen

- Falls mehrere Fehlermeldungen angezeigt werden sollen, kann ein Array von Fehlern verwendet werden.
- Da die Validierung in `onBlur` stattfindet (also **nach** dem initialen Render), muss das Ergebnis in State gespeichert werden. Nur wenn State aktualisiert wird, löst React einen Re-render aus — und nur dann werden die Fehlermeldungen im JSX angezeigt. Ein einfaches `const errors = []` würde bei jedem Render neu (leer) erstellt und hätte keinen Effekt auf die Anzeige.

```typescript
import { useState } from 'react'

export function InputComponent() {
    const [value, setValue] = useState('')
    const [errors, setErrors] = useState<string[]>([])
    const handleChange = (event: React.ChangeEvent<HTMLInputElement>) => {
        setValue(event.target.value)
    }

    const handleBlur = () => {
        const newErrors: string[] = []
        if (value.length < 5) {
            newErrors.push('Der Wert muss mindestens 5 Zeichen lang sein.')
        }
        if (!value.includes('@')) {
            newErrors.push('Der Wert muss ein @-Zeichen enthalten.')
        }
        setErrors(newErrors)
    }
    return (
        <div>
            <input type="text" value={value} onChange={handleChange} onBlur={handleBlur} />
            {errors.map((error, index) => (
                <div key={index}>{error}</div>
            ))}
        </div>
    )
}
```
